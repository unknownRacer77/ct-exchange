    <script>
        let tradeData = [];
        let itemDbData = [];
        let filteredTrade = [];
        let filteredItemDb = [];
        let priceChartInstance = null;
        let currentTradePage = 1;
        let currentItemDbPage = 1;

        function getItemsPerPage() {
            return window.innerWidth <= 768 ? 10 : 28;
        }

        function switchView(viewName) {
            document.querySelectorAll('.view').forEach(el => el.classList.remove('active-view'));
            document.querySelectorAll('.nav-tab-btn').forEach(el => el.classList.remove('active'));
            
            document.getElementById(`view-${viewName}`).classList.add('active-view');
            document.getElementById(`tab-${viewName}`).classList.add('active');
        }

        async function fetchAndRenderTrade() {
            try {
                const res = await fetch('ct_trade_db.json?t=' + new Date().getTime());
                const lastModifiedHeader = res.headers.get('Last-Modified');
                const data = await res.json();

                let syncTimeStr = '';
                let isGolden = false;
                let isFokju = false;

                if (!Array.isArray(data) && data.items) {
                    tradeData = data.items;
                    syncTimeStr = data.updatedAt || data.updated_at || data.syncTime || '';
                    isGolden = !!(data.golden || data.isGolden);
                    isFokju = !!(data.fokju || data.isFokju);
                } else {
                    tradeData = data;
                    if (lastModifiedHeader) {
                        const dateObj = new Date(lastModifiedHeader);
                        syncTimeStr = dateObj.toLocaleTimeString('ko-KR', { hour: '2-digit', minute: '2-digit', second: '2-digit' });
                    }
                    const now = new Date();
                    const hour = now.getHours();
                    isGolden = (hour >= 20 && hour < 22);
                    isFokju = (hour >= 21 && hour < 22);
                }

                document.getElementById('sync-time').innerText = syncTimeStr ? `동기화: ${syncTimeStr}` : '동기화: 완료';
                document.getElementById('golden-badge').classList.toggle('active', isGolden);
                document.getElementById('fokju-badge').classList.toggle('active', isFokju);

            } catch (e) {
                tradeData = [];
                document.getElementById('sync-time').innerText = '동기화: 파일 없음';
            }
            applyTradeFilters();
        }

        function applyTradeFilters() {
            const keyword = document.getElementById('search-input').value.trim().toLowerCase();
            const sort = document.getElementById('sort-select').value;

            filteredTrade = tradeData.filter(item => (item.itemName || item.name || '').toLowerCase().includes(keyword));

            if (sort === 'price-asc') filteredTrade.sort((a, b) => a.price - b.price);
            if (sort === 'price-desc') filteredTrade.sort((a, b) => b.price - a.price);

            document.getElementById('total-count').innerText = `총 ${filteredTrade.length}개 매물`;
            currentTradePage = 1;
            renderTradeList();
        }

        function renderTradeList() {
            const listEl = document.getElementById('item-list');
            if (filteredTrade.length === 0) {
                listEl.innerHTML = `<div class="loading">등록된 매물이 없습니다.</div>`;
                document.getElementById('trade-pagination').innerHTML = '';
                return;
            }

            const itemsPerPage = getItemsPerPage();
            const totalPages = Math.ceil(filteredTrade.length / itemsPerPage);
            if (currentTradePage > totalPages) currentTradePage = totalPages;

            const start = (currentTradePage - 1) * itemsPerPage;
            const pageItems = filteredTrade.slice(start, start + itemsPerPage);

            listEl.innerHTML = pageItems.map(item => {
                const name = item.itemName || item.name;
                const iconSrc = item.icon || item.image || item.img || item.iconPath || '';
                return `
                    <div class="item-card" onclick="openModal('${name}')">
                        <div class="item-middle">
                            ${iconSrc ? `<img class="item-thumbnail" src="${iconSrc}" onerror="this.style.display='none'">` : `<div class="item-thumbnail" style="display:flex;align-items:center;justify-content:center;color:#64748b;font-size:11px;">No Img</div>`}
                            <div class="item-title">${name}</div>
                        </div>
                        <div class="item-footer">
                            <span class="time-text">${item.time || ''}</span>
                            <span class="price-text">${Number(item.price).toLocaleString()} CT</span>
                        </div>
                    </div>
                `;
            }).join('');

            renderTradePagination(totalPages);
        }

        function renderTradePagination(totalPages) {
            const pageEl = document.getElementById('trade-pagination');
            if (totalPages <= 1) {
                pageEl.innerHTML = '';
                return;
            }

            const maxButtons = 10;
            let startPage = Math.floor((currentTradePage - 1) / maxButtons) * maxButtons + 1;
            let endPage = Math.min(startPage + maxButtons - 1, totalPages);

            let html = '';
            if (startPage > 1) {
                html += `<button class="page-btn" onclick="changeTradePage(${startPage - 1})">‹</button>`;
            }
            for (let i = startPage; i <= endPage; i++) {
                html += `<button class="page-btn ${i === currentTradePage ? 'active' : ''}" onclick="changeTradePage(${i})">${i}</button>`;
            }
            if (endPage < totalPages) {
                html += `<button class="page-btn" onclick="changeTradePage(${endPage + 1})">›</button>`;
            }
            pageEl.innerHTML = html;
        }

        function changeTradePage(page) {
            currentTradePage = page;
            renderTradeList();
            window.scrollTo({ top: 0, behavior: 'smooth' });
        }

        async function initItemDatabase() {
            try {
                const res = await fetch('ctr_items_db.json?t=' + new Date().getTime());
                const data = await res.json();
                itemDbData = (!Array.isArray(data) && data.items) ? data.items : data;
            } catch (err) {
                itemDbData = [];
            }
            filteredItemDb = itemDbData;
            initDropdowns(itemDbData);
            filterItemTable();
        }

        function initDropdowns(data) {
            const cSel = document.getElementById('filter-class');
            const pSel = document.getElementById('filter-parts');
            const sSel = document.getElementById('filter-shop');

            const classSet = [...new Set(data.map(i => getItemClass(i)).filter(c => c && c !== '-'))].sort();
            const partsSet = [...new Set(data.map(i => getVal(i, ['parts', 'part', 'partName', '파츠'])).filter(Boolean))].sort();
            const shopSet = [...new Set(data.map(i => getVal(i, ['shopgroup', 'shopGroup', 'shop', '상점그룹'])).filter(Boolean))].sort();

            cSel.innerHTML = '<option value="">전체 Class</option>' + classSet.map(c => `<option value="${c}">Class ${c}</option>`).join('');
            pSel.innerHTML = '<option value="">전체 파츠</option>' + partsSet.map(p => `<option value="${p}">${p}</option>`).join('');
            sSel.innerHTML = '<option value="">전체 상점그룹</option>' + shopSet.map(s => `<option value="${s}">${s}</option>`).join('');
        }

        function filterItemTable() {
            const nameKey = document.getElementById('item-search').value.toLowerCase();
            const classKey = document.getElementById('filter-class').value;
            const partsKey = document.getElementById('filter-parts').value;
            const shopKey = document.getElementById('filter-shop').value;

            filteredItemDb = itemDbData.filter(item => {
                const itemCls = getItemClass(item);
                const itemName = (getVal(item, ['name', 'itemName', '이름']) || '').toLowerCase();
                const itemParts = getVal(item, ['parts', 'part', 'partName', '파츠']) || '';
                const itemShop = getVal(item, ['shopgroup', 'shopGroup', 'shop', '상점그룹']) || '';
                return (!nameKey || itemName.includes(nameKey)) &&
                       (!classKey || itemCls === classKey) &&
                       (!partsKey || itemParts === partsKey) &&
                       (!shopKey || itemShop === shopKey);
            });

            currentItemDbPage = 1;
            renderItemTable();
        }

        function getVal(item, keys) {
            for (let k of keys) {
                if (item[k] !== undefined && item[k] !== null && item[k] !== '' && item[k] !== '-') return item[k];
            }
            return null;
        }

      function getItemClass(item) {
            const val = getVal(item, ['carclass', 'class', 'carClass', 'itemClass', 'Class', 'CLASS', 'car_class', 'item_class', 'vehicleClass', 'grade', 'Grade', 'rank', 'Rank', '클래스', '등급']);
            return val ? String(val).trim().toUpperCase() : '-';
        }

        function getItemTp(item) {
            const val = getVal(item, ['tp', 'TP', 'tPoint', 'tpoint', 't_point', 'point', 'Point', '점수', '포인트']);
            return (val !== null && val !== undefined && String(val).trim() !== '') ? val : '-';
        }

        function renderItemTable() {
            const tbody = document.getElementById('item-db-list');
            const pageEl = document.getElementById('pagination');

            if (filteredItemDb.length === 0) {
                tbody.innerHTML = `<tr><td colspan="6" class="loading">데이터가 없습니다.</td></tr>`;
                pageEl.innerHTML = '';
                return;
            }

            const itemsPerPage = 10;
            const totalPages = Math.ceil(filteredItemDb.length / itemsPerPage);
            if (currentItemDbPage > totalPages) currentItemDbPage = totalPages;

            const start = (currentItemDbPage - 1) * itemsPerPage;
            const pageItems = filteredItemDb.slice(start, start + itemsPerPage);

            tbody.innerHTML = pageItems.map(item => {
                const cls = getItemClass(item);
                const tp = getItemTp(item);
                
                let iconSrc = getVal(item, ['image', 'icon', 'Icon', 'img', 'Img', 'iconPath', 'thumbnail', 'thumb', '아이콘', '이미지']) || '';
                if (iconSrc.startsWith('/')) iconSrc = '.' + iconSrc;

                let priceVal = getVal(item, ['price', 'Price', 'PRICE', 'cost', 'Cost', 'sellPrice', 'sell_price', 'shopPrice', 'shop_price', 'money', 'Money', 'ct', 'CT', 'amount', 'Amount', '가격', '금액']);
                let cleanPrice = priceVal ? String(priceVal).replace(/,/g, '').trim() : '';
                const priceStr = (cleanPrice !== '' && !isNaN(Number(cleanPrice))) 
                                 ? Number(cleanPrice).toLocaleString() + ' CT' : '-';

                return `
                    <tr>
                        <td>${iconSrc ? `<img src="${iconSrc}" style="width:28px;height:28px;object-fit:contain;" onerror="this.onerror=null; this.parentNode.innerHTML='🖼️';">` : '🖼️'}</td>
                        <td>${getVal(item, ['name', 'itemName', '이름']) || '-'}</td>
                        <td>${getVal(item, ['parts', 'part', 'partName', '파츠']) || '-'}</td>
                        <td><span class="class-badge ${cls}">${cls}</span></td>
                        <td>${tp}</td>
                        <td>${priceStr}</td>
                    </tr>
                `;
            }).join('');

            renderItemDbPagination(totalPages);
        }

        function renderItemDbPagination(totalPages) {
            const pageEl = document.getElementById('pagination');
            if (totalPages <= 1) {
                pageEl.innerHTML = '';
                return;
            }

            const maxButtons = 10;
            let startPage = Math.floor((currentItemDbPage - 1) / maxButtons) * maxButtons + 1;
            let endPage = Math.min(startPage + maxButtons - 1, totalPages);

            let html = '';
            if (startPage > 1) {
                html += `<button class="page-btn" onclick="changeItemDbPage(${startPage - 1})">‹</button>`;
            }
            for (let i = startPage; i <= endPage; i++) {
                html += `<button class="page-btn ${i === currentItemDbPage ? 'active' : ''}" onclick="changeItemDbPage(${i})">${i}</button>`;
            }
            if (endPage < totalPages) {
                html += `<button class="page-btn" onclick="changeItemDbPage(${endPage + 1})">›</button>`;
            }
            pageEl.innerHTML = html;
        }

        function changeItemDbPage(page) {
            currentItemDbPage = page;
            renderItemTable();
        }

        function openModal(itemName) {
            const sameItems = tradeData.filter(i => (i.itemName || i.name) === itemName);
            const prices = sameItems.map(i => Number(i.price)).filter(p => !isNaN(p));

            let minP = prices.length ? Math.min(...prices) : 0;
            let maxP = prices.length ? Math.max(...prices) : 0;
            let medP = 0;

            if (prices.length) {
                const sorted = [...prices].sort((a, b) => a - b);
                const mid = Math.floor(sorted.length / 2);
                medP = sorted.length % 2 !== 0 ? sorted[mid] : Math.round((sorted[mid - 1] + sorted[mid]) / 2);
            }

            const html = `
                <h3 style="color:#f8fafc; margin-bottom:12px;">${itemName}</h3>
                <div class="modal-grid">
                    <div class="info-box">
                        <span>매물 수량</span>
                        <strong>${sameItems.length}개</strong>
                    </div>
                    <div class="info-box">
                        <span>최저가</span>
                        <strong>${minP > 0 ? minP.toLocaleString() + ' CT' : '-'}</strong>
                    </div>
                    <div class="info-box">
                        <span>중앙값</span>
                        <strong>${medP > 0 ? medP.toLocaleString() + ' CT' : '-'}</strong>
                    </div>
                    <div class="info-box">
                        <span>최고가</span>
                        <strong>${maxP > 0 ? maxP.toLocaleString() + ' CT' : '-'}</strong>
                    </div>
                </div>
                <div class="chart-section">
                    <h4>등록 매물 가격 분포</h4>
                    <div class="chart-wrapper">
                        <canvas id="priceChart"></canvas>
                    </div>
                </div>
            `;

            document.getElementById('modal-body').innerHTML = html;
            document.getElementById('modal').style.display = 'flex';

            setTimeout(() => { renderPriceChart(prices); }, 50);
        }

        function renderPriceChart(prices) {
            const ctx = document.getElementById('priceChart');
            if (!ctx) return;

            if (priceChartInstance) {
                priceChartInstance.destroy();
            }

            const sortedPrices = [...prices].sort((a, b) => a - b);
            const chartLabels = sortedPrices.map((_, idx) => `매물 ${idx + 1}`);

            priceChartInstance = new Chart(ctx, {
                type: 'line',
                data: {
                    labels: chartLabels.length > 0 ? chartLabels : ['매물 없음'],
                    datasets: [{
                        label: '가격 (CT)',
                        data: sortedPrices.length > 0 ? sortedPrices : [0],
                        borderColor: '#38bdf8',
                        backgroundColor: 'rgba(56, 189, 248, 0.15)',
                        fill: true,
                        tension: 0.3,
                        pointRadius: 4,
                        pointBackgroundColor: '#facc15'
                    }]
                },
                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    plugins: { legend: { display: false } },
                    scales: {
                        x: { ticks: { color: '#64748b', font: { size: 10 } }, grid: { display: false } },
                        y: { ticks: { color: '#64748b', font: { size: 10 } }, grid: { color: '#1e293b' } }
                    }
                }
            });
        }

        function closeModal() {
            document.getElementById('modal').style.display = 'none';
        }
