    <script>
        // 화면 전환 로직
        function switchView(viewId) {
            document.querySelectorAll('.view').forEach(v => v.classList.remove('active-view'));
            document.querySelectorAll('.nav-tab-btn').forEach(b => b.classList.remove('active'));

            document.getElementById('view-' + viewId).classList.add('active-view');
            document.getElementById('tab-' + viewId).classList.add('active');

            if (viewId === 'items' && itemData.length === 0) {
                initItemDatabase();
            }
        }

        // --- [PART 1] 거래소 검색 로직 ---
        let rawItems = [];

        async function fetchAndRenderTrade() {
            const listEl = document.getElementById('item-list');
            const syncEl = document.getElementById('sync-time');

            try {
                const res = await fetch('data.json?t=' + new Date().getTime());
                if (!res.ok) throw new Error('data.json 로드 실패');
                
                const jsonData = await res.json();
                rawItems = [];
                jsonData.forEach(p => {
                    if (p.items && Array.isArray(p.items)) {
                        rawItems.push(...p.items);
                    }
                });

                syncEl.innerText = `동기화: ${jsonData[0].timestamp}`;
                applyTradeFilters();
            } catch (err) {
                console.error(err);
                listEl.innerHTML = `<div class="loading" style="color: #f43f5e;">데이터를 불러올 수 없습니다.</div>`;
            }
        }

        function applyTradeFilters() {
            const query = document.getElementById('search-input').value.toLowerCase();
            const sortVal = document.getElementById('sort-select').value;

            let filtered = rawItems.filter(item => {
                const nameMatch = item.name && item.name.toLowerCase().includes(query);
                const nickMatch = item.nickname && item.nickname.toLowerCase().includes(query);
                return nameMatch || nickMatch;
            });

            if (sortVal === 'price-asc') {
                filtered.sort((a, b) => parsePrice(a.price) - parsePrice(b.price));
            } else if (sortVal === 'price-desc') {
                filtered.sort((a, b) => parsePrice(b.price) - parsePrice(a.price));
            }

            document.getElementById('total-count').innerText = `총 ${filtered.length}개 매물`;
            renderTradeList(filtered);
        }

        function parsePrice(str) {
            if (!str) return 0;
            return parseInt(str.toString().replace(/[^0-9]/g, '')) || 0;
        }

        function renderTradeList(items) {
            const listEl = document.getElementById('item-list');
            if (!items || items.length === 0) {
                listEl.innerHTML = `<div class="loading">등록된 매물이 없습니다.</div>`;
                return;
            }

            listEl.innerHTML = items.map(item => {
                const priceStr = parsePrice(item.price) > 0 ? parsePrice(item.price).toLocaleString() + ' CT' : '-';
                const imagePath = item.image_path ? item.image_path + '?t=' + new Date().getTime() : '';

                return `
                    <div class="item-card">
                        <div class="item-card-top">
                            <span class="seller-id">ID: ${item.nickname || '알 수 없음'}</span>
                        </div>
                        <div class="item-middle">
                            <img src="${imagePath}" class="item-thumbnail" alt="아이템" onerror="this.style.display='none'">
                            <div class="item-title">${item.name || '알 수 없음'}</div>
                        </div>
                        <div class="item-footer">
                            <span class="time-text">⏱ ${item.time || '-'}</span>
                            <span class="price-text">${priceStr}</span>
                        </div>
                    </div>
                `;
            }).join('');
        }

        // --- [PART 2] 아이템 정보 검색 로직 ---
        let itemData = [], filteredItemData = [], currentItemPage = 1;
        const itemsPerPage = 20;

        async function initItemDatabase() {
            try {
                const res = await fetch('ctr_items_db.json?t=' + new Date().getTime());
                itemData = await res.json();
                filteredItemData = itemData;
                initDropdowns(itemData);
                filterItemTable();
            } catch (err) {
                console.error(err);
                document.getElementById('item-db-list').innerHTML = `<tr><td colspan="6" class="loading" style="color: #f43f5e;">DB 로드 실패</td></tr>`;
            }
        }

        function getItemClass(item) {
            return String(item.carclass || item.class || '').trim().toUpperCase();
        }

        function getItemTp(item) {
            if (item.tp !== undefined && item.tp !== null && item.tp !== '') return item.tp;
            if (item.details) {
                for (let [k, v] of Object.entries(item.details)) {
                    if (k.toUpperCase().includes('TP')) return v;
                }
            }
            return '-';
        }

        function getClassBadgeHtml(cls) {
            if (!cls) return '-';
            return ['C', 'B', 'A', 'S', 'G', 'R'].includes(cls) ? `<span class="class-badge ${cls}">${cls}</span>` : cls;
        }

        function initDropdowns(data) {
            const classSet = [...new Set(data.map(i => getItemClass(i)))].filter(Boolean).sort();
            const partsSet = [...new Set(data.map(i => i.parts))].filter(Boolean).sort();
            const shopSet = [...new Set(data.map(i => i.shopgroup))].filter(Boolean).sort();

            const cSel = document.getElementById('filter-class');
            const pSel = document.getElementById('filter-parts');
            const sSel = document.getElementById('filter-shop');

            classSet.forEach(c => cSel.innerHTML += `<option value="${c}">Class ${c}</option>`);
            partsSet.forEach(p => pSel.innerHTML += `<option value="${p}">${p}</option>`);
            shopSet.forEach(s => sSel.innerHTML += `<option value="${s}">${s}</option>`);
        }

        function filterItemTable() {
            const keyword = document.getElementById('item-search').value.toLowerCase();
            const sc = document.getElementById('filter-class').value;
            const sp = document.getElementById('filter-parts').value;
            const ss = document.getElementById('filter-shop').value;

            filteredItemData = itemData.filter(item => {
                return item.name.toLowerCase().includes(keyword) &&
                       (!sc || getItemClass(item) === sc) &&
                       (!sp || item.parts === sp) &&
                       (!ss || item.shopgroup === ss);
            });
            currentItemPage = 1;
            renderItemTable(filteredItemData);
        }

        function renderItemTable(data) {
            const tbody = document.getElementById('item-db-list');
            tbody.innerHTML = '';
            
            const start = (currentItemPage - 1) * itemsPerPage;
            const pageData = data.slice(start, start + itemsPerPage);

            if (pageData.length === 0) {
                tbody.innerHTML = `<tr><td colspan="6" class="loading">검색 결과가 없습니다.</td></tr>`;
                document.getElementById('pagination').innerHTML = '';
                return;
            }

            pageData.forEach(item => {
                const cls = getItemClass(item);
                const tr = document.createElement('tr');
                tr.onclick = () => showModal(item);
                tr.innerHTML = `
                    <td><img src="https://ctr.domi.kr${item.image}" width="28" height="28" onerror="this.style.display='none'"></td>
                    <td style="text-align:left; padding-left:10px; font-weight:600;">${item.name}</td>
                    <td>${item.parts || '-'}</td>
                    <td>${getClassBadgeHtml(cls)}</td>
                    <td>${getItemTp(item)}</td>
                    <td>${item.price || '-'}</td>
                `;
                tbody.appendChild(tr);
            });
            renderItemPagination(data.length);
        }

        // 수정된 페이지네이션 (가시성을 위해 최대 5개 번호와 이전/다음 버튼만 표시)
        function renderItemPagination(totalItems) {
            const container = document.getElementById('pagination');
            container.innerHTML = '';
            const pageCount = Math.ceil(totalItems / itemsPerPage);
            if (pageCount <= 1) return;

            const maxVisible = 5;
            let start = Math.max(1, currentItemPage - Math.floor(maxVisible / 2));
            let end = Math.min(pageCount, start + maxVisible - 1);
            if (end - start + 1 < maxVisible) {
                start = Math.max(1, end - maxVisible + 1);
            }

            // 이전 버튼
            if (currentItemPage > 1) {
                const prevBtn = document.createElement('button');
                prevBtn.className = 'page-btn';
                prevBtn.innerText = '◀';
                prevBtn.onclick = () => { currentItemPage--; renderItemTable(filteredItemData); };
                container.appendChild(prevBtn);
            }

            // 페이지 번호 버튼
            for (let i = start; i <= end; i++) {
                const btn = document.createElement('button');
                btn.className = `page-btn ${i === currentItemPage ? 'active' : ''}`;
                btn.innerText = i;
                btn.onclick = () => {
                    currentItemPage = i;
                    renderItemTable(filteredItemData);
                };
                container.appendChild(btn);
            }

            // 다음 버튼
            if (currentItemPage < pageCount) {
                const nextBtn = document.createElement('button');
                nextBtn.className = 'page-btn';
                nextBtn.innerText = '▶';
                nextBtn.onclick = () => { currentItemPage++; renderItemTable(filteredItemData); };
                container.appendChild(nextBtn);
            }
        }

        function showModal(item) {
            const body = document.getElementById('modal-body');
            const cls = getItemClass(item);
            const tpVal = getItemTp(item);

            let html = `
                <div style="display:flex; align-items:center; gap:12px; margin-bottom:12px;">
                    <img src="https://ctr.domi.kr${item.image}" width="42" height="42" onerror="this.style.display='none'" style="border:1px solid #28354d; border-radius:6px; padding:2px; background:#0a0f1d;">
                    <div style="font-size:15px; font-weight:bold; color:#fff;">${item.name}</div>
                </div>
                <div class="tooltip-card">
                    <div style="font-size:13px; font-weight:bold; color:#fff; margin-bottom:4px;">${item.name}</div>
                    <div class="stat-green">장착위치 ${item.parts || ''}</div>
            `;

            if (item.details) {
                for (let [k, v] of Object.entries(item.details)) {
                    const strV = String(v);
                    if (strV.startsWith('+')) html += `<div class="stat-green">${k}${strV}</div>`;
                    else if (strV.startsWith('-')) html += `<div class="stat-red">${k}${strV}</div>`;
                }
            }
            if (tpVal !== '-') html += `<div class="stat-yellow" style="margin-top:2px;">요구 TP ${tpVal}</div>`;
            if (item.description) html += `<div style="margin-top:6px; color:#9ca3af; font-size:11px; border-top:1px solid #28354d; padding-top:4px;">${item.description}</div>`;
            html += `</div><table class="info-table" style="font-size:12px;"><tr><th>상세 정보</th><th>값</th></tr>`;

            const baseInfo = [
                ['가격', item.price],
                ['적용차량', getClassBadgeHtml(cls)],
                ['상점그룹', item.shopgroup],
                ['거래소 분류', item.parts]
            ];
            baseInfo.forEach(([k, v]) => { if (v) html += `<tr><td>${k}</td><td>${v}</td></tr>`; });

            if (item.details) {
                for (let [k, v] of Object.entries(item.details)) {
                    let styleClass = String(v).startsWith('+') ? 'class="stat-green"' : String(v).startsWith('-') ? 'class="stat-red"' : '';
                    html += `<tr><td>${k}</td><td ${styleClass}>${v}</td></tr>`;
                }
            }
            html += `</table>`;
            body.innerHTML = html;
            document.getElementById('modal').style.display = 'flex';
        }

            document.getElementById('modal').style.display = 'none';
